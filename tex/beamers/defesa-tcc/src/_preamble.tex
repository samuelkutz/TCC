% --- motivo de ondas (Boussinesq) para as faixas azuis ------------------
% Usar dentro de um tikzpicture com remember picture/overlay; recorta na
% faixa azul (metade inferior) e desenha tres senoides quase transparentes.
% O argumento opcional e a defasagem (em graus) das senoides, para que
% cada slide de faixa azul tenha um recorte de onda diferente.
\newcommand{\tccwaves}[1][0]{%
  \begin{scope}
    \clip ([yshift=15pt]current page.west) rectangle (current page.south east);
    \begin{scope}[shift={(current page.south west)},x=1cm,y=1cm,white,
                  smooth,variable=\x,domain=-0.3:16.3,samples=160,line width=1.1pt]
      \draw[opacity=0.17] plot ({\x},{1.55+0.20*sin(\x*52+(#1))});
      \draw[opacity=0.11] plot ({\x},{1.12+0.26*sin(\x*44+120+1.4*(#1))});
      \draw[opacity=0.07] plot ({\x},{0.68+0.22*sin(\x*60+240+0.7*(#1))});
    \end{scope}
  \end{scope}%
}

% --- slide divisor de secao ------------------------------------------
\setbeamertemplate{section page}{%
  \begin{tikzpicture}[remember picture,overlay]
    \fill[white] ([yshift=15pt]current page.west) rectangle (current page.north east);
    \fill[structure] ([yshift=15pt]current page.west) rectangle (current page.south east);
    \tccwaves[\the\value{section}*71]
    \node[anchor=east, font=\huge\bfseries, text width=0.85\linewidth, align=right]
      at ([yshift=-0.30\paperheight,xshift=-25pt]current page.north east) (section)
      {\textcolor{structure}{\insertsection}};
  \end{tikzpicture}%
}
\AtBeginSection[]{%
  {%
    \setbeamertemplate{background}{}%
    \begin{frame}[plain,noframenumbering]
      \sectionpage
    \end{frame}%
  }%
}
