  \begin{tikzpicture}[remember picture,overlay]
    \fill[white] ([yshift=15pt]current page.west) rectangle (current page.north east);
    \fill[structure] ([yshift=15pt]current page.west) rectangle (current page.south east);
    \node[anchor=east, font=\huge\bfseries, text width=0.85\linewidth, align=right]
      at ([yshift=-0.30\paperheight,xshift=-25pt]current page.north east) (section)
      {\textcolor{structure}{\insertsection}};
  \end{tikzpicture}%
}

  {%
    \setbeamertemplate{background}{}%

  }%
