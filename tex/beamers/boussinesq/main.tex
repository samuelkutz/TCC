\section{Introdução}

\begin{frame}[t]
    \frametitle{Introdução}
    As Equações Diferenciais Parciais (EDPs) são uma das principais ferramentas para descrever fenômenos da natureza, e em muitos problemas, precisamos encontrar uma solução numérica por métodos clássicos discretos (diferenças finitas).
    \vspace{0.5cm}

    Este trabalho investiga as \textbf{\color{UniBlue}Redes Neurais Fisicamente Guiadas (PINNs)}, uma alternativa baseada em \textit{Deep Learning} que aproxima soluções de forma contínua, incorporando as leis físicas (EDPs) diretamente na função de perda da rede neural \parencite{raissi2017}.
    \vspace{1cm}

    \begin{block}{Objetivo}
        Analisar a eficácia e os limites da abordagem PINN na solução do \textbf{sistema de Boussinesq}, um modelo clássico da hidrodinâmica que descreve a dinâmica de ondas aquáticas.
    \end{block}
\end{frame}

\begin{frame}[t]
    \frametitle{Redes Neurais Fisicamente Guiadas (PINNs)}
    \begin{columns}[T,totalwidth=\textwidth]
        \begin{column}{0.55\textwidth}
            As PINNs visam solucionar um sistema genérico, em termos de um operador \\ diferencial $\mathcal{D}$:
            \begin{equation*}
                \left\{
                \begin{aligned}
                    & \mathcal{D}[u](x,t) = 0, \quad \forall (x,t) \in \Omega \times (0, T] \\
                    & u(x,0) = u_0(x), \quad x \in \Omega
                \end{aligned}
                \right.
            \end{equation*}
            O treinamento minimiza o \textbf{Erro Quadrático Médio (EQM)} dos resíduos:
            \begin{block}{Função de Perda}
                \resizebox{\textwidth}{!}{
                $
                \text{EQM}(\theta) = \underbrace{\frac{1}{N_{\mathcal{D}}} \sum_{i=1}^{N_{\mathcal{D}}} \left( \mathcal{D}[u_\theta](x_i, t_i) \right)^2}_{\text{Resíduo da EDP}} +
                                   \underbrace{\frac{1}{N_{0}} \sum_{j=1}^{N_{0}} \left( u_\theta(x_j, 0) - u_0(x_j) \right)^2}_{\text{Cond. Inicial}}
                $
                }
            \end{block}
        \end{column}
        \begin{column}{0.45\textwidth}
            A ideia é treinar uma rede neural $u_{\theta}(x,t)$ para satisfazer:
            \begin{itemize}
                \item A EDP no interior do domínio;
                \item As condições iniciais.
            \end{itemize}
            \begin{figure}
                \includegraphics[width=0.85\linewidth]{redeneural_theta.png}
                \caption*{\footnotesize{Arquitetura MLP que mapeia \\ $(x,t) \to u_{\theta}(x,t)$.}}
            \end{figure}
        \end{column}
    \end{columns}
\end{frame}

\section{Metodologia}

\begin{frame}[t]
    \frametitle{O Problema Físico: Sistema de Boussinesq}
    Este sistema descreve a propagação de ondas em águas rasas, modelando a evolução da superfície, $\eta(x,t)$, e a média vertical da velocidade horizontal, $u(x,t)$:
    
    \begin{block}{Sistema de Boussinesq e Condições Iniciais}
        $
        \left\{
        \begin{aligned}
            &\eta_t + u_x + \alpha(\eta u)_x = 0, \\
            &u_t + \eta_x + \alpha u u_x - \tfrac{\beta}{3} u_{xxt} = 0,
        \end{aligned}
        \right.
        $
        com $\eta(x, 0) = \text{sech}^2(x)$ e $u(x, 0) = 0$.
    \end{block}

    \begin{block}{Caso Limite: $\alpha = \beta = 0$ (Equação de Onda Linear)}
        Neste caso, o problema admite uma solução exata (fórmula de d'Alembert), onde a onda inicial se divide em duas:

        $$ \eta(x,t) = \frac{1}{2}\text{sech}^2(x-t) + \frac{1}{2}\text{sech}^2(x+t) $$
    \end{block}
    
    Os parâmetros $\alpha$ e $\beta$ controlam os efeitos \textbf{não lineares} e \textbf{dispersivos}.
\end{frame}

\begin{frame}[t]
    \frametitle{Amostragem Adaptativa (RAR)}
    Uma dificuldade conhecida em PINNs é o \textbf{viés espectral} \cite{rahaman2019spectral}: a tendência da rede em aprender primeiro as componentes de baixa frequência da solução.
    
    Para enfrentar este desafio, foi implementada a amostragem adaptativa \textit{Residual-based Adaptive Refinement} (RAR) \parencite{wu2023comprehensive}, que prioriza pontos em regiões de \textbf{maior erro}.
    
    \begin{block}{Ideia Central do RAR}
       Amostrar novos pontos de treinamento com base em uma densidade de probabilidade $p(x,t)$ que considera o resíduo da EDP, $\epsilon(x,t) = |\mathcal{D}[u_\theta](x,t)|$:
       \begin{equation*}
        p(x,t) \propto \frac{|\epsilon(x,t)|^k}{\mathbb{E}[|\epsilon(x,t)|^k]} + c.
       \end{equation*}
       Para $k, c \gte 0$, forçando a rede a focar onde a física é mais complexa e o erro é maior.
    \end{block}
\end{frame}

\begin{frame}[t]
    \frametitle{Detalhes da Implementação}
    \begin{itemize}
            \vspace{0.5cm}
        \item \textbf{Framework:} O modelo foi desenvolvido utilizando a biblioteca \textbf{PyTorch}.
        \vspace{0.5cm}
        \item \textbf{Arquitetura:} Uma rede neural do tipo \textit{Multilayer Perceptron} (MLP) com funções de ativação tangente hiperbólica (tanh).
        \vspace{0.5cm}
        \item \textbf{Otimização:} O treinamento foi realizado com o otimizador \textbf{Adam}, seguido por L-BFGS para ajuste fino.
        \vspace{0.5cm}
        \item \textbf{Hardware:} Os experimentos foram executados em GPUs NVIDIA para acelerar o processo de treinamento.
    \end{itemize}
\end{frame}

\section{Conclusões}

\begin{frame}[t]
    \frametitle{Conclusões e Considerações Finais}
    Este trabalho demonstrou que as PINNs possuem um desempenho de \textbf{dupla face}:
    \vspace{0.4cm}
    \begin{itemize}
        \item Eficazes em regimes de baixa complexidade (soluções suaves/simples).
        \item Em regimes de complexidade intermediária ($\alpha, \beta$ significativos), a performance é limitada pelo viés espectral.
    \end{itemize}
    \vspace{0.4cm}
    
    A amostragem adaptativa (RAR) se mostrou uma \textbf{estratégia útil para mitigar este efeito}, mas não o elimina por completo.
    \vspace{0.4cm}
    
    Isso reforça que as PINNs são uma ferramenta emergente, e não uma substituta universal para métodos clássicos, necessitando de mais estudo sobre seus limites.
    \vspace{0.4cm}
    
\end{frame}

\begin{frame}[t]
    \frametitle{Próximos Passos: E se não conhecemos a EDP?}
    Uma crítica comum ao \textit{Physics-Informed ML} é que conhecer a EDP exata de um sistema é, muitas vezes, inviável.
    \vspace{0.3cm}

    Como próximo passo, investigaremos uma abordagem puramente baseada em dados: os \textbf{Fourier Neural Operators (FNOs)} \parencite{li2020}. FNOs aprendem mapeamentos entre espaços de funções, sendo capazes de modelar sistemas físicos sem conhecimento explícito das equações governantes.

    \begin{center}
        \begin{minipage}{0.9\textwidth} % Controla a largura dos blocos
            \begin{block}{Estrutura de um Operador Neural}
                $$G_\theta = Q \circ \sigma(W_L + \mathcal{K}_L) \circ \dots \circ \sigma(W_1 + \mathcal{K}_1) \circ P$$
            \end{block}
            
            \begin{block}{Kernel Convolucional do FNO (para a camada $\ell$)}
                 $$
                 (\mathcal{K}v_{\ell-1})(x) = \int_{\Omega} \kappa(x-y)v_{\ell-1}(y)dy = \mathcal{F}^{-1}\big( \mathcal{F}(\kappa) \cdot \mathcal{F}(v_{\ell-1}) \big)(x)
                 $$
            \end{block}
        \end{minipage}
    \end{center}
\end{frame}
